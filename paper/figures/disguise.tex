% How a subject model is disguised before the agent sees it (Figure 2). Input from main.tex; needs tikz. Same design
% language as figures/overview.tex; the four hidden neurons are tracked by colour through the permutation.
\providecolor{aaink}{HTML}{111111}
\providecolor{aamuted}{HTML}{6B6B6B}
\providecolor{aarule}{HTML}{E3E3E3}
\definecolor{nA}{HTML}{7F4BF3}
\definecolor{nB}{HTML}{F2C94C}
\definecolor{nC}{HTML}{30C695}
\definecolor{nD}{HTML}{F2754C}
\begin{tikzpicture}[
    font=\sffamily\footnotesize, text=aaink, >={Stealth[length=1.8mm]},
    card/.style={draw=aarule, fill=white, line width=0.6pt, rounded corners=7pt},
    head/.style={font=\rmfamily\normalsize, anchor=west, text=aaink},
    sq/.style={rectangle, fill=aaink, minimum size=0.22cm, inner sep=0pt},
    note/.style={font=\sffamily\scriptsize, text=aamuted, align=center},
    lab/.style={font=\sffamily\scriptsize, text=aaink},
    go/.style={-{Stealth[length=2.2mm, width=2.4mm]}, line width=1.1pt, draw=aamuted!60},
  ]
  % one MLP as two matrices: W_in (rows = hidden neurons), W_out (columns = hidden neurons)
  % #1 x-origin, #2 order of the four neurons (colour names)
  \newcommand{\mlp}[2]{%
    \foreach \c [count=\k from 0] in {#2} {
      \fill[\c!80, rounded corners=0.8pt] (#1, 3.35 - 0.36*\k) rectangle ++(1.5, 0.3);
      \fill[\c!80, rounded corners=0.8pt] (#1 + 2.05 + 0.36*\k, 1.95) rectangle ++(0.3, 1.7);
    }
    \draw[aarule, line width=0.5pt, rounded corners=2pt] (#1 - 0.04, 2.25) rectangle (#1 + 1.54, 3.69);
    \draw[aarule, line width=0.5pt, rounded corners=2pt] (#1 + 2.01, 1.91) rectangle (#1 + 3.49, 3.69);
    \node[lab, anchor=north] at (#1 + 0.75, 2.18) {$\mathsf{W}_{\mathsf{in}}$};
    \node[lab, anchor=north] at (#1 + 2.75, 1.84) {$\mathsf{W}_{\mathsf{out}}$};
  }

  \draw[card] (0, 0) rectangle (4.6, 4.75);
  \draw[card] (5.15, 0) rectangle (9.75, 4.75);
  \draw[card] (10.3, 0) rectangle (16, 4.75);
  \draw[go] (4.64, 2.6) -- (5.11, 2.6);
  \draw[go] (9.79, 2.6) -- (10.26, 2.6);

  % ---------------------------------------------------------------- 1. original checkpoint
  \node[sq] (s1) at (0.42, 4.33) {};
  \node[head] at ([xshift=3pt]s1.east) {Public checkpoint};
  \mlp{0.55}{nA,nB,nC,nD}
  \node[note] at (2.3, 0.85) {hidden neurons 1--4, in order;\\named tensors and a config file};

  % ---------------------------------------------------------------- 2. permuted
  \node[sq] (s2) at (5.57, 4.33) {};
  \node[head] at ([xshift=3pt]s2.east) {Permuted};
  \mlp{5.7}{nC,nA,nD,nB}
  \node[note] at (7.45, 0.85) {rows of $\mathsf{W}_{\mathsf{in}}$ and columns of $\mathsf{W}_{\mathsf{out}}$\\move together: same function};

  % ---------------------------------------------------------------- 3. what the agent receives
  \node[sq] (s3) at (10.72, 4.33) {};
  \node[head] at ([xshift=3pt]s3.east) {What the agent sees};
  \node[lab, anchor=west, text width=5.1cm] at (10.6, 2.75) {%
    {\color{aamuted}\textbullet}\ tensors renamed to canonical names\\[2pt]
    {\color{aamuted}\textbullet}\ configuration stripped\\[2pt]
    {\color{aamuted}\textbullet}\ MLP neurons and attention heads permuted at random, per model\\[2pt]
    {\color{aamuted}\textbullet}\ the vocabulary, to read tokens};
  \node[note, text width=5.2cm] at (13.15, 0.85) {matching tensors against public files\\no longer identifies the model};
\end{tikzpicture}
