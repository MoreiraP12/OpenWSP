% How a subject model is disguised before the agent sees it (Figure 2). Input from main.tex; needs tikz.
% Colours match figures/overview.tex; the four hidden neurons are tracked by colour through the permutation.
\definecolor{cink}{HTML}{2B2B2B}
\definecolor{cpanel}{HTML}{F3F4F6}
\definecolor{nA}{HTML}{0072B2}
\definecolor{nB}{HTML}{E69F00}
\definecolor{nC}{HTML}{009E73}
\definecolor{nD}{HTML}{CC79A7}
\begin{tikzpicture}[
    font=\sffamily\footnotesize, text=cink, >={Stealth[length=1.8mm]},
    panel/.style={fill=cpanel, rounded corners=6pt},
    head/.style={font=\sffamily\bfseries\small, anchor=west},
    note/.style={font=\sffamily\scriptsize, text=black!60, align=center},
    lab/.style={font=\sffamily\scriptsize, text=cink},
    go/.style={->, line width=1.6pt, draw=black!35},
  ]
  % one MLP as two matrices: W_in (rows = hidden neurons), W_out (columns = hidden neurons)
  % #1 x-origin, #2 order of the four neurons (colour names)
  \newcommand{\mlp}[2]{%
    \foreach \c [count=\k from 0] in {#2} {
      % row k of W_in
      \fill[\c!75] (#1, 3.35 - 0.36*\k) rectangle ++(1.5, 0.3);
      % column k of W_out
      \fill[\c!75] (#1 + 2.05 + 0.36*\k, 1.95) rectangle ++(0.3, 1.7);
    }
    \draw[black!35, line width=0.4pt] (#1, 2.27) rectangle (#1 + 1.5, 3.65);
    \draw[black!35, line width=0.4pt] (#1 + 2.05, 1.95) rectangle (#1 + 3.49, 3.65);
    \node[lab, anchor=north] at (#1 + 0.75, 2.2) {$\mathsf{W}_{\mathsf{in}}$};
    \node[lab, anchor=north] at (#1 + 2.77, 1.88) {$\mathsf{W}_{\mathsf{out}}$};
  }

  \fill[panel] (0, 0) rectangle (4.6, 4.75);
  \fill[panel] (5.15, 0) rectangle (9.75, 4.75);
  \fill[panel] (10.3, 0) rectangle (16, 4.75);
  \draw[go] (4.62, 2.6) -- (5.13, 2.6);
  \draw[go] (9.77, 2.6) -- (10.28, 2.6);

  % ---------------------------------------------------------------- 1. original checkpoint
  \node[head] at (0.3, 4.35) {Public checkpoint};
  \mlp{0.55}{nA,nB,nC,nD}
  \node[note] at (2.3, 0.85) {hidden neurons 1--4, in order;\\named tensors and a config file};

  % ---------------------------------------------------------------- 2. permuted
  \node[head] at (5.45, 4.35) {Permuted};
  \mlp{5.7}{nC,nA,nD,nB}
  \node[note] at (7.45, 0.85) {rows of $\mathsf{W}_{\mathsf{in}}$ and columns of $\mathsf{W}_{\mathsf{out}}$\\move together: same function};

  % ---------------------------------------------------------------- 3. what the agent receives
  \node[head] at (10.6, 4.35) {What the agent sees};
  \node[lab, anchor=west, text width=5.1cm] at (10.6, 2.75) {%
    \textbullet\ tensors renamed to canonical names\\[2pt]
    \textbullet\ configuration stripped\\[2pt]
    \textbullet\ MLP neurons and attention heads permuted at random, per model\\[2pt]
    \textbullet\ the vocabulary, to read tokens};
  \node[note, text width=5.2cm] at (13.15, 0.85) {matching tensors against public files\\no longer identifies the model};
\end{tikzpicture}
