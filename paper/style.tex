% Design layer of the WSP-Bench paper, in the style of the Artificial Analysis site: a modern serif for text and
% headings, a neo-grotesque sans for labels and tables, the same palette and lab colours as the figures
% (scripts/make_figures.py), card-like tables and abstract. \input from main.tex after the packages; deleting the
% \input line gives back the plain article look (the tables then need booktabs only).

% ---- headings: regular-weight serif; a section starts with a small black square, as on the site
\usepackage{titlesec}
\titleformat{\section}{\rmfamily\fontsize{17}{21}\selectfont\color{aaink}}%
  {\raisebox{0.12em}{\rule{0.42em}{0.42em}}\hspace{0.55em}{\color{aamuted}\thesection}}{0.5em}{}
\titlespacing*{\section}{0pt}{20pt plus 4pt minus 2pt}{8pt plus 2pt}
\titleformat{\subsection}{\rmfamily\fontsize{13}{16}\selectfont\color{aaink}}{{\color{aamuted}\thesubsection}}{0.6em}{}
\titlespacing*{\subsection}{0pt}{14pt plus 3pt minus 2pt}{5pt plus 2pt}
\titleformat{\paragraph}[runin]{\sffamily\small\bfseries\color{aaink}}{}{0pt}{}
\titlespacing*{\paragraph}{0pt}{7pt plus 2pt minus 1pt}{0.6em}

% ---- captions: sans label, serif text
\captionsetup{font=small, labelfont={sf,bf}, labelsep=period, justification=justified, singlelinecheck=false}
\captionsetup[table]{skip=6pt}
